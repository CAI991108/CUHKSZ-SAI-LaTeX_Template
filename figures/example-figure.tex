\documentclass[tikz,border=8pt]{standalone}
\usetikzlibrary{arrows.meta,positioning}
\begin{document}
\begin{tikzpicture}[
    box/.style={draw, rounded corners=2pt, align=center,
                minimum width=3.0cm, minimum height=1.2cm, font=\small},
    arrow/.style={-{Latex[length=2mm]}, thick}
]
    \node[box] (input) {Documents\\and metadata};
    \node[box, right=1.6cm of input] (model) {Classifier +\\claim extractor};
    \node[box, right=1.6cm of model] (review) {Human review\\and decision};
    \node[box, below=1.6cm of model] (log) {Evidence log\\and audit trail};

    \draw[arrow] (input) -- (model);
    \draw[arrow] (model) -- (review);
    \draw[arrow] (review) -- ++(0,-0.9) -| (log);
    \draw[arrow] (log) -- (model);
